import * as rx_callbacks from '../src/rx.ores';
import class Observable from '../src/rx.ores';

// Keep the mutable callback accumulator outside the benchmark's loop body.
// This exercises legal owned lexical capture without implicitly binding self.
pub fnc pipeline_checksum(): int {
  val Observable<int> source = rx_callbacks.from_values([1, 2, 3, 4, 5, 6, 7, 8]);
  val Fnc<int, int> twice = |value| -> {
    val int item = value as int;
    return item * 2;
  };
  val Observable<int> doubled = rx_callbacks.map(source, twice);
  val Fnc<int, bool> above_four = |value| -> {
    val int item = value as int;
    return item > 4;
  };
  val Observable<int> selected = rx_callbacks.filter(doubled, above_four);
  val Observable<int> limited = rx_callbacks.take(selected, 3);

  let int emitted_checksum = 0;
  val Fnc<int, bool> sink = |value| -> {
    val int item = value as int;
    emitted_checksum = emitted_checksum + item;
    return true;
  };
  // Upstream consumption (5) is not the terminal emitted-value sum (24).
  val int consumed = rx_callbacks.for_each(limited, sink);
  if consumed != 5; then
    val Result<bool, String> failure = Err("callback benchmark consumed unexpected source count");
    failure.expect("rx callback benchmark");
  fi
  return emitted_checksum;
}

pub fnc main(): void {
  let int checksum = 0;
  for let iteration = 0; iteration < 20000; iteration++ do
    checksum = checksum + pipeline_checksum();
  done
  stdio.println(checksum);
  return;
}
